--[[
    Utils / BoneLegAdapter  2.1.0-roots
 
    Ноги: RightLegTarget / LeftLegTarget + RightKneePole / LeftKneePole
    Руки: RightHandTarget / LeftHandTarget + RightElbowPole / LeftElbowPole
 
    НОВОЕ (2.1.0): RootControl-хэндлы для корней цепочек
      RightLegRootControl / LeftLegRootControl
      RightArmRootControl / LeftArmRootControl
    Двигаешь хэндл -> корневая кость (RightBoneRoot, LeftLegBoneRoot, ...)
    переезжает в эту точку (Bone.Transform получает смещение),
    а IK-цепочка продолжает тянуться к таргету и смотреть на pole.
    Если хэндла нет, поведение такое же, как в 2.0.0.
 
    НОВОЕ (2.2.0): зависимость от спины (Utils/BoneSpineAdapter)
      ноги -> основа считается относительно TorsoBoneRoot
      руки -> основа считается относительно TorsoBone.003
    Если спина не активна, поведение как в 2.1.0.
    RootControl-хэндл по-прежнему задаёт ДОПОЛНИТЕЛЬНОЕ смещение корня
    от родного (непозированного) места.

    Двигаются только Bone.Transform (root и mid кости).
    MeshPart, Motor6D, WeldConstraint не изменяются.
]]

local LEG_ROOT_HINTS = { "thigh", "upper", "hip", "femur", "root" }
local LEG_MID_HINTS = { "knee", "shin", "lower", "calf", "tibia" }
local LEG_END_HINTS = { "ankle", "foot", "toe" }

local ARM_ROOT_HINTS = { "shoulder", "upper", "arm", "root" }
local ARM_MID_HINTS = { "elbow", "forearm", "lower", "arm" }
local ARM_END_HINTS = { "hand", "wrist", "palm", "finger" }

local LEG_WORDS = { "leg", "thigh", "knee", "shin", "foot", "ankle", "calf" }
local ARM_WORDS = { "arm", "shoulder", "elbow", "forearm", "wrist", "hand" }
local function attachWithMotor6D(part0, part1, motorName)
	if not part0 or not part1 then 
		warn("Ошибка: part0 или part1 не существуют!")
		return nil 
	end
	local motor = Instance.new("Motor6D")
	motor.Name = motorName or "Motor6D"
	motor.Part0 = part0 -- Основная деталь (родоначальник движения)
	motor.Part1 = part1 -- Привязываемая деталь

	-- Настраиваем начальное смещение на основе текущего положения деталей в пространстве
	motor.C0 = part0.CFrame:Inverse() * part1.CFrame

	-- Помещаем Motor6D внутрь Part0 (можно заменить на part1 или модель персонажа)
	motor.Parent = part0

	return motor
end
local function lower(text)
	return string.lower(text)
end

local function containsAny(text, hints)
	for _, hint in hints do
		if string.find(text, hint, 1, true) then
			return true
		end
	end
	return false
end

local function strContains(text, word)
	return string.find(text, word, 1, true) ~= nil
end

local function unit(vector)
	if not vector or vector.Magnitude < 0.0001 then
		return nil
	end
	return vector.Unit
end

local function safeWarn(message)
	warn("[BoneLegAdapter] " .. tostring(message))
end

local function getRoot(rig)
	local humanoid = rig:FindFirstChildOfClass("Humanoid")
	if humanoid and humanoid.RootPart then
		return humanoid.RootPart
	end
	if rig.PrimaryPart then
		return rig.PrimaryPart
	end
	for _, descendant in rig:GetDescendants() do
		if descendant:IsA("BasePart") and descendant.Name == "HumanoidRootPart" then
			return descendant
		end
	end
	return rig:FindFirstChildWhichIsA("BasePart", true)
end
module.GetRootPart = getRoot

-- Контролы лежат в rig.IK (специальная модель), а не прямо в риге.
-- Ищем: rig.IK -> прямо в риге -> рекурсивно по всему ригу.
local function findControl(rig, name)
	if not name then
		return nil
	end

	local part = nil
	local ikModel = rig:FindFirstChild("IK")
	if ikModel then
		part = ikModel:FindFirstChild(name)
	end
	if not part then
		part = rig:FindFirstChild(name)
	end
	if not part then
		part = rig:FindFirstChild(name, true)
	end

	if part and part:IsA("BasePart") then
		return part
	end
	return nil
end

local function allBones(rig)
	local list = {}
	for _, descendant in rig:GetDescendants() do
		if descendant:IsA("Bone") then
			table.insert(list, descendant)
		end
	end
	return list
end

local function findBone(rig, name)
	if not name or name == "" then
		return nil
	end
	for _, bone in allBones(rig) do
		if bone.Name == name then
			return bone
		end
	end
	return nil
end

local function pathText(bone, depth)
	local parts = {}
	local current = bone
	local count = 0

	while current and count < (depth or 4) do
		table.insert(parts, lower(current.Name))
		current = current.Parent
		count += 1
	end

	return table.concat(parts, "/")
end

local function regionOf(bone)
	local current = bone
	local count = 0

	while current and count < 6 do
		local text = lower(current.Name)
		if containsAny(text, ARM_WORDS) then
			return "Arm"
		end
		if containsAny(text, LEG_WORDS) then
			return "Leg"
		end
		current = current.Parent
		count += 1
	end

	return nil
end

local function sideOf(bone)
	local current = bone
	local count = 0

	while current and count < 5 do
		local text = lower(current.Name)
		if strContains(text, "right") then
			return "Right"
		end
		if strContains(text, "left") then
			return "Left"
		end
		current = current.Parent
		count += 1
	end

	return nil
end

local function findMesh(rig, bone, config)
	local current = bone.Parent
	while current do
		if current:IsA("MeshPart") then
			return current
		end
		if current:IsA("BasePart") then
			local nested = current:FindFirstChildWhichIsA("MeshPart", true)
			if nested then
				return nested
			end
		end
		current = current.Parent
	end

	if config and config.Mesh then
		local named = rig:FindFirstChild(config.Mesh, true)
		if named and named:IsA("MeshPart") then
			return named
		end
	end

	return nil
end

local function findEndBone(mid, kind)
	local hints = kind == "Arm" and ARM_END_HINTS or LEG_END_HINTS

	for _, child in mid:GetChildren() do
		if child:IsA("Bone") and containsAny(lower(child.Name), hints) then
			return child
		end
	end

	return nil
end

local function scorePair(parentBone, childBone, sideName, kind)
	local region = regionOf(parentBone) or regionOf(childBone)
	if region and region ~= kind then
		return -math.huge
	end

	local parentName = lower(parentBone.Name)
	local childName = lower(childBone.Name)
	local path = pathText(parentBone, 5) .. "/" .. pathText(childBone, 3)
	local side = lower(sideName)

	local score = region == kind and 20 or 0

	if kind == "Leg" then
		if containsAny(parentName, LEG_ROOT_HINTS) then score += 8 end
		if containsAny(childName, LEG_MID_HINTS) then score += 12 end
	else
		if containsAny(parentName, ARM_ROOT_HINTS) then score += 8 end
		if containsAny(childName, ARM_MID_HINTS) then score += 12 end
	end

	if strContains(path, side) then score += 6 end
	if strContains(parentName, side) then score += 4 end
	if strContains(childName, side) then score += 4 end
	if parentBone.Parent and parentBone.Parent:IsA("BasePart") then score += 2 end

	return score
end

local function detectChain(rig, sideName, kind, claimed, warnings)
	local candidates = {}

	for _, parentBone in allBones(rig) do
		for _, child in parentBone:GetChildren() do
			if child:IsA("Bone") then
				local score = scorePair(parentBone, child, sideName, kind)
				if score > 8 then
					table.insert(candidates, {
						Root = parentBone,
						Mid = child,
						Score = score,
						Path = pathText(parentBone, 5),
					})
				end
			end
		end
	end

	table.sort(candidates, function(a, b)
		return a.Score > b.Score
	end)

	for _, candidate in candidates do
		if not claimed[candidate.Root] and not claimed[candidate.Mid] then
			if strContains(lower(candidate.Path), lower(sideName)) then
				claimed[candidate.Root] = kind
				claimed[candidate.Mid] = kind
				return candidate.Root, candidate.Mid
			end
		end
	end

	for _, candidate in candidates do
		if not claimed[candidate.Root] and not claimed[candidate.Mid] then
			claimed[candidate.Root] = kind
			claimed[candidate.Mid] = kind
			return candidate.Root, candidate.Mid
		end
	end

	local bones = {}
	for _, bone in allBones(rig) do
		table.insert(bones, bone:GetFullName())
	end

	table.insert(
		warnings,
		string.format(
			"%s %s: не найдена цепочка. Кости рига: %s",
			kind,
			sideName,
			#bones > 0 and table.concat(bones, ", ") or "НЕТ"
		)
	)

	return nil
end

local function estimateLength(mesh, rootPosition, midPosition, direction)
	local localAxis = mesh.CFrame:VectorToObjectSpace(direction)
	local half = (
		math.abs(localAxis.X) * mesh.Size.X
			+ math.abs(localAxis.Y) * mesh.Size.Y
			+ math.abs(localAxis.Z) * mesh.Size.Z
	) * 0.5

	local centerProjection = (mesh.CFrame.Position - rootPosition):Dot(direction)
	local midProjection = (midPosition - rootPosition):Dot(direction)
	local farProjection = centerProjection + half

	return math.max(farProjection - midProjection, 0.2)
end

local function captureChain(rig, sideName, kind, config, claimed, warnings)
	local root = findBone(rig, config.Root)
	local mid = findBone(rig, config.Mid)

	if not root or not mid then
		local detectedRoot, detectedMid = detectChain(rig, sideName, kind, claimed, warnings)
		root = root or detectedRoot
		mid = mid or detectedMid
	end

	if not root or not mid then
		return nil
	end

	claimed[root] = claimed[root] or kind
	claimed[mid] = claimed[mid] or kind

	local mesh = findMesh(rig, root, config)
	if not mesh then
		table.insert(warnings, kind .. " " .. sideName .. ": не найден MeshPart для костей")
		return nil
	end

	local savedRoot = root.Transform
	local savedMid = mid.Transform
	root.Transform = CFrame.new()
	mid.Transform = CFrame.new()

	local rootWorld = root.WorldCFrame
	local midWorld = mid.WorldCFrame
	local rootPosition = rootWorld.Position
	local midPosition = midWorld.Position
	local l1 = (midPosition - rootPosition).Magnitude

	root.Transform = savedRoot
	mid.Transform = savedMid

	if l1 < 0.01 then
		table.insert(warnings, kind .. " " .. sideName .. ": длина первой кости 0")
		return nil
	end

	local direction = unit(midPosition - rootPosition) or Vector3.new(0, -1, 0)
	local endBone = findBone(rig, config.End) or findEndBone(mid, kind)
	local endPosition
	local l2

	if endBone then
		endPosition = endBone.WorldCFrame.Position
		l2 = (endPosition - midPosition).Magnitude
	else
		l2 = config.Length or estimateLength(mesh, rootPosition, midPosition, direction)
		endPosition = midPosition + direction * l2
	end

	if l2 < 0.01 then
		table.insert(warnings, kind .. " " .. sideName .. ": длина второй кости 0")
		return nil
	end

	local meshCFrame = mesh.CFrame

	return {
		Kind = kind,
		Side = sideName,
		Name = sideName .. kind,

		Root = root,
		Mid = mid,
		EndBone = endBone,
		Mesh = mesh,

		L1 = l1,
		L2 = l2,

		LocalRoot = meshCFrame:PointToObjectSpace(rootPosition),
		LocalMid = meshCFrame:PointToObjectSpace(midPosition),
		LocalEnd = meshCFrame:PointToObjectSpace(endPosition),

		RootRestRotation = meshCFrame:ToObjectSpace(rootWorld).Rotation,
		MidRestRotation = meshCFrame:ToObjectSpace(midWorld).Rotation,

		RootBindWorld = rootPosition,
		MidBindWorld = midPosition,
		EndBindWorld = endPosition,

		Target = nil,
		Pole = nil,
		RootControl = nil,
	}
end

local function configFor(chain)
	return chain.Kind == "Arm" and CONFIG.Arms[chain.Side] or CONFIG.Legs[chain.Side]
end

-- Текущая "родная" позиция корня (следует за мешем, если риг двигали)
local function restRootWorld(chain)
	return chain.Mesh.CFrame:PointToWorldSpace(chain.LocalRoot)
end

local function bind(rig)
	local warnings = {}
	local claimed = {}
	local chains = {}

	local legRight = captureChain(rig, "Right", "Leg", CONFIG.Legs.Right, claimed, warnings)
	local legLeft = captureChain(rig, "Left", "Leg", CONFIG.Legs.Left, claimed, warnings)

	if legRight then table.insert(chains, legRight) end
	if legLeft then table.insert(chains, legLeft) end

	local armRight = captureChain(rig, "Right", "Arm", CONFIG.Arms.Right, claimed, warnings)
	local armLeft = captureChain(rig, "Left", "Arm", CONFIG.Arms.Left, claimed, warnings)

	if armRight then table.insert(chains, armRight) end
	if armLeft then table.insert(chains, armLeft) end

	local meta = {
		Rig = rig,
		Chains = chains,
		Warnings = warnings,
		Legs = { Right = legRight, Left = legLeft },
		Arms = { Right = armRight, Left = armLeft },
		Enabled = #chains > 0,
	}

	for _, chain in chains do
		local config = chain.Kind == "Arm" and CONFIG.Arms[chain.Side] or CONFIG.Legs[chain.Side]
		chain.Target = findControl(rig, config.Target)
		chain.Pole = findControl(rig, config.Pole)
		chain.RootControl = config.RootControl and findControl(rig, config.RootControl) or nil
	end

	cache[rig] = meta
	return meta
end

function module.Bind(rig)
	if not rig or not rig:IsA("Model") then
		return nil
	end
	return cache[rig] or bind(rig)
end

function module.Rebind(rig)
	cache[rig] = nil
	return bind(rig)
end

function module.IsBoneLegRig(rig)
	if not rig or not rig:IsA("Model") then
		return false
	end
	if not rig:FindFirstChildOfClass("Humanoid") then
		return false
	end
	return module.Bind(rig).Enabled
end

function module.Restore(rig)
	local meta = module.Bind(rig)
	if not meta then
		return
	end

	for _, chain in meta.Chains do
		if chain.Root.Parent then chain.Root.Transform = CFrame.new() end
		if chain.Mid.Parent then chain.Mid.Transform = CFrame.new() end
	end
end

function module.GetChains(rig)
	local meta = module.Bind(rig)
	if not meta then
		return {}
	end
	return meta.Chains
end

function module.GetSides(rig)
	local list = {}
	for _, chain in module.GetChains(rig) do
		table.insert(list, {
			Name = chain.Name,
			Kind = chain.Kind,
			Side = chain.Side,
			Thigh = chain.Root,
			Knee = chain.Mid,
			Mesh = chain.Mesh,
			Target = chain.Target,
			Pole = chain.Pole,
			RootControl = chain.RootControl,
			L1 = chain.L1,
			L2 = chain.L2,
		})
	end
	return list
end

function module.GetTargets(rig)
	local list = {}
	local seen = {}

	local function add(part)
		if part and part:IsA("BasePart") and not seen[part] then
			seen[part] = true
			table.insert(list, part)
		end
	end

	for _, chain in module.GetChains(rig) do
		add(chain.Target)
		add(chain.Pole)
		add(chain.RootControl)
	end

	return list
end

-- Только хэндлы корней (RightLegRootControl, LeftLegRootControl, ...)
function module.GetRootControls(rig)
	local list = {}

	for _, chain in module.GetChains(rig) do
		local part = chain.RootControl
		if not part or not part.Parent then
			local config = configFor(chain)
			part = config.RootControl and findControl(rig, config.RootControl) or nil
			chain.RootControl = part
		end
		if part and part:IsA("BasePart") then
			table.insert(list, part)
		end
	end

	return list
end

-- Вернуть хэндлы корней в родное положение (корень кости на месте)
function module.ResetRootControls(rig)
	local count = 0

	for _, chain in module.GetChains(rig) do
		local part = chain.RootControl
		if part and part.Parent and chain.Mesh.Parent then
			part.CFrame = CFrame.new(restRootWorld(chain))
			count += 1
		end
	end

	return count
end

local function ensureControl(rig, name, position, color, size, shape)
	local IkModel 
	if rig:FindFirstChild("IK") then
		IkModel=rig:FindFirstChild("IK")
	else
		IkModel = Instance.new("Model",rig)
		IkModel.Name = "IK"
		local HM = Instance.new("Humanoid",IkModel)
		HM.Name = "Humanoid"
		local IKRoot = rig.HumanoidRootPart:Clone()
		IKRoot.Parent = IkModel
		IkModel.PrimaryPart =IKRoot
	end
	local existing = IkModel:FindFirstChild(name)

	if existing then
		if not existing:IsA("BasePart") then
			return nil, name .. " занято объектом " .. existing.ClassName
		end
		existing.Anchored = true
		existing.CanCollide = false
		existing.CanTouch = false
		existing.CanQuery = false
		existing.CastShadow = false
		existing.Massless = true
		return existing
	end

	local part = Instance.new("Part")
	part.Name = name
	part.Shape = shape or Enum.PartType.Ball
	part.Size = Vector3.new(size, size, size)
	part.Material = Enum.Material.Neon
	part.Color = color
	part.Transparency = 0.15
	part.Anchored = true
	part.CanCollide = false
	part.CanTouch = false
	part.CanQuery = false
	part.CastShadow = false
	part.Massless = true
	part.Locked = false
	part.CFrame = CFrame.new(position)
	part.Parent = IkModel
	addLabel(part)
	if autopodviazka ==true then

		attachWithMotor6D(IkModel.HumanoidRootPart,part,part.Name)
	end
	return part
end

function module.EnsureControls(rig)
	if not rig or not rig:IsA("Model") then
		return false, "выделен не Model"
	end
	if not rig:FindFirstChildOfClass("Humanoid") then
		return false, "в риге нет Humanoid"
	end

	local meta = module.Rebind(rig)
	if not meta.Enabled then
		return false, "кости не найдены. " .. table.concat(meta.Warnings, " | ")
	end

	module.Restore(rig)

	local rootPart = getRoot(rig)
	local forward = rootPart and rootPart.CFrame.LookVector or Vector3.new(0, 0, -1)
	local rightVector = rootPart and rootPart.CFrame.RightVector or Vector3.new(1, 0, 0)
	local created = 0

	for _, chain in meta.Chains do
		local isArm = chain.Kind == "Arm"
		local config = isArm and CONFIG.Arms[chain.Side] or CONFIG.Legs[chain.Side]

		local target, targetError = ensureControl(
			rig,
			config.Target,
			chain.EndBindWorld,
			isArm and CONFIG.TargetColor.Arm or CONFIG.TargetColor.Leg,
			CONFIG.TargetSize
		)

		if not target then
			return false, targetError
		end

		local sideSign = chain.Side == "Right" and 1 or -1
		local polePosition

		if isArm then
			polePosition = chain.MidBindWorld
			- forward * math.max(chain.L1 * 0.7, 0.5)
				+ rightVector * (sideSign * math.max(chain.L1 * 0.35, 0.3))
		else
			polePosition = chain.MidBindWorld
				+ forward * math.max(chain.L1 * 0.7, 0.5)
		end

		local pole, poleError = ensureControl(
			rig,
			config.Pole,
			polePosition,
			isArm and CONFIG.PoleColor.Arm or CONFIG.PoleColor.Leg,
			CONFIG.PoleSize,
			Enum.PartType.Cylinder
		)

		if not pole then
			return false, poleError
		end

		if not chain.Target then created += 1 end
		if not chain.Pole then created += 1 end

		chain.Target = target
		chain.Pole = pole

		-- RootControl: хэндл корневой кости (RightBoneRoot и т.п.)
		if CONFIG.RootControlsEnabled and config.RootControl then
			local hadRootControl = chain.RootControl ~= nil

			local rootControl, rootError = ensureControl(
				rig,
				config.RootControl,
				restRootWorld(chain),
				isArm and CONFIG.RootColor.Arm or CONFIG.RootColor.Leg,
				CONFIG.RootSize,
				Enum.PartType.Block
			)

			if not rootControl then
				return false, rootError
			end

			if not hadRootControl then created += 1 end
			chain.RootControl = rootControl
		end
	end

	local tag = CONFIG.Tag
	if shared._IKRigGlobal and shared._IKRigGlobal.IKRig_Tag then
		tag = shared._IKRigGlobal.IKRig_Tag
	end

	if not rig:HasTag(tag) then
		rig:AddTag(tag)
	end

	if not rig.PrimaryPart and rootPart then
		rig.PrimaryPart = rootPart
	end

	rig:SetAttribute("BoneIKReady", true)

	print("[Bone IK] контролы готовы, новых объектов: " .. created)
	for _, chain in meta.Chains do
		print(string.format(
			"  %s %s / %s | target=%s | pole=%s | rootControl=%s",
			chain.Kind,
			chain.Root.Name,
			chain.Mid.Name,
			chain.Target and chain.Target.Name or "MISSING",
			chain.Pole and chain.Pole.Name or "MISSING",
			chain.RootControl and chain.RootControl.Name or "off"
			))
	end

	for _, warning in meta.Warnings do
		warn("[Bone IK] " .. warning)
	end

	return true, "готово: " .. #meta.Chains .. " цепочек"
end

local function swing(fromDirection, toDirection)
	local from = unit(fromDirection)
	local to = unit(toDirection)

	if not from or not to then
		return CFrame.new()
	end

	local dot = math.clamp(from:Dot(to), -1, 1)

	if dot > 0.99999 then
		return CFrame.new()
	end

	if dot < -0.99999 then
		local axis = from:Cross(Vector3.xAxis)
		if axis.Magnitude < 0.0001 then
			axis = from:Cross(Vector3.zAxis)
		end
		return CFrame.fromAxisAngle(axis.Unit, math.pi)
	end

	return CFrame.fromAxisAngle(from:Cross(to).Unit, math.acos(dot))
end

-- Дельта якорной кости спины (BoneSpineAdapter). nil = спина не активна.
local function getAnchorDelta(rig, chain)
	local global = shared._IKRigGlobal
	local spine = global and global.utils and global.utils.BoneSpineAdapter
	if not spine then
		return nil
	end

	local anchorName = spine.GetAnchorName(chain.Kind)
	return spine.GetDelta(rig, anchorName)
end

local function solveChain(rig, chain)
	local target = chain.Target
	if not target or not target.Parent then
		local config = chain.Kind == "Arm" and CONFIG.Arms[chain.Side] or CONFIG.Legs[chain.Side]
		target = findControl(rig, config.Target)
		chain.Target = target
	end
	if not target then
		return
	end

	local pole = chain.Pole
	if not pole or not pole.Parent then
		local config = chain.Kind == "Arm" and CONFIG.Arms[chain.Side] or CONFIG.Legs[chain.Side]
		pole = findControl(rig, config.Pole)
		chain.Pole = pole
	end

	local rootControl = nil
	if CONFIG.RootControlsEnabled then
		rootControl = chain.RootControl
		if not rootControl or not rootControl.Parent then
			local config = configFor(chain)
			rootControl = config.RootControl and findControl(rig, config.RootControl) or nil
			chain.RootControl = rootControl
		end
	end

	if not chain.Root.Parent or not chain.Mid.Parent or not chain.Mesh.Parent then
		return
	end

	local meshCFrame = chain.Mesh.CFrame
	local restRoot = meshCFrame:PointToWorldSpace(chain.LocalRoot)
	local restMid = meshCFrame:PointToWorldSpace(chain.LocalMid)
	local restEnd = meshCFrame:PointToWorldSpace(chain.LocalEnd)

	-- Родная (непозированная) ориентация кости. Transform считается относительно неё.
	local unposedRootPosition = restRoot
	local unposedRootRotation = meshCFrame.Rotation * chain.RootRestRotation

	-- Смещение корня: куда пользователь утащил RootControl относительно родного места.
	-- Считается от НЕпозированного места, поэтому хэндл на родном месте = "следуй за спиной".
	local rootOffset = Vector3.zero
	if rootControl and rootControl:IsA("BasePart") then
		rootOffset = rootControl.CFrame.Position - unposedRootPosition
		if rootOffset.Magnitude < CONFIG.RootEpsilon then
			rootOffset = Vector3.zero
		end
	end

	local rootRestRotation = unposedRootRotation
	local midRestRotation = meshCFrame.Rotation * chain.MidRestRotation

	-- Основа цепочки зависит от спины: ноги -> TorsoBoneRoot, руки -> TorsoBone.003.
	-- Вся rest-цепочка переезжает вместе с якорной костью (длины костей не меняются).
	local anchorDelta = getAnchorDelta(rig, chain)
	if anchorDelta then
		restRoot = anchorDelta:PointToWorldSpace(restRoot)
		restMid = anchorDelta:PointToWorldSpace(restMid)
		restEnd = anchorDelta:PointToWorldSpace(restEnd)
		rootRestRotation = anchorDelta.Rotation * rootRestRotation
		midRestRotation = anchorDelta.Rotation * midRestRotation
	end

	local rootPosition = restRoot

	if rootOffset ~= Vector3.zero then
		rootPosition = restRoot + rootOffset
		restMid = restMid + rootOffset
		restEnd = restEnd + rootOffset
	end

	local l1 = (restMid - rootPosition).Magnitude
	local l2 = (restEnd - restMid).Magnitude

	if l1 < 0.01 or l2 < 0.01 then
		return
	end

	local toTarget = target.CFrame.Position - rootPosition
	local rawDistance = toTarget.Magnitude

	if rawDistance < 0.0001 then
		return
	end

	local direction = toTarget / rawDistance
	local distance = math.clamp(rawDistance, math.abs(l1 - l2) + 0.001, l1 + l2 - 0.001)

	local bend = nil

	if pole and pole.Parent then
		local toPole = pole.CFrame.Position - rootPosition
		bend = toPole - direction * toPole:Dot(direction)
	end

	if not bend or bend.Magnitude < 0.001 then
		local rootPart = getRoot(rig)
		local fallbackForward = rootPart and rootPart.CFrame.LookVector or meshCFrame.LookVector
		if chain.Kind == "Arm" then
			fallbackForward = -fallbackForward
		end
		bend = fallbackForward - direction * fallbackForward:Dot(direction)
	end

	if bend.Magnitude < 0.001 then
		bend = direction:Cross(Vector3.yAxis)
	end
	if bend.Magnitude < 0.001 then
		bend = meshCFrame.RightVector
	end

	bend = bend.Unit

	local along = (l1 * l1 - l2 * l2 + distance * distance) / (2 * distance)
	local height = math.sqrt(math.max(l1 * l1 - along * along, 0))

	local solvedMid = rootPosition + direction * along + bend * height
	local solvedEnd = rootPosition + direction * distance

	local delta1 = swing(restMid - rootPosition, solvedMid - rootPosition)

	-- Поворот вокруг (уже сдвинутого) корня + перенос корня в точку RootControl.
	-- Смещение переводим в локальные оси кости: Final = Rest * Transform.
	-- rootRestRotation уже включает позу спины, unposedRootRotation — родная ориентация кости.
	local rootRotation = unposedRootRotation:Inverse() * delta1 * rootRestRotation
	local localOffset = unposedRootRotation:VectorToObjectSpace(rootPosition - unposedRootPosition)

	chain.Root.Transform = CFrame.new(localOffset) * rootRotation

	local inheritedEnd = delta1 * (restEnd - restMid)
	local delta2 = swing(inheritedEnd, solvedEnd - solvedMid)

	chain.Mid.Transform =
		midRestRotation:Inverse()
		* (delta1:Inverse() * delta2 * delta1)
		* midRestRotation
end

function module.SolveRig(rig)
	local meta = module.Bind(rig)
	if not meta or not meta.Enabled then
		return false
	end

	for _, chain in meta.Chains do
		local ok, err = pcall(solveChain, rig, chain)
		if not ok then
			safeWarn("solver " .. chain.Name .. ": " .. tostring(err))
		end
	end

	return true
end

function module.GetSample(rig)
	local meta = module.Bind(rig)
	local sample = {}

	if not meta then
		return sample
	end

	for _, chain in meta.Chains do
		if chain.Root.Parent then
			sample[chain.Root.Name] = chain.Root.Transform
		end
		if chain.Mid.Parent then
			sample[chain.Mid.Name] = chain.Mid.Transform
		end
	end

	return sample
end

function module.DumpBones(rig)
	print("[Bone IK] кости рига " .. rig:GetFullName())
	for _, bone in allBones(rig) do
		print(string.format(
			"  %-42s parent=%-20s region=%s side=%s",
			bone:GetFullName(),
			bone.Parent and bone.Parent.Name or "nil",
			tostring(regionOf(bone)),
			tostring(sideOf(bone))
			))
	end
end

return module
