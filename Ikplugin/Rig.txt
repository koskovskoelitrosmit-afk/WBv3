local function SafeUnit(v: Vector3)
	if v.Magnitude == 0 then return Vector3.zero end
	return v.Unit
end

local module = {}

local function isBoneRig(rig)
	return BoneLegAdapter.IsBoneLegRig(rig)
end


----/// RIG VALIDATION
function module.ValidateRig(rig: Model): boolean
	if not rig then return false end
	if not rig:IsA("Model") then return false end
	if not rig:FindFirstChildOfClass("Humanoid") then return false end
	if module.IsBoneRig(rig) then
		return BoneLegAdapter.GetRootPart(rig) ~= nil
	end
	if not rig.PrimaryPart then return false end
	return true
end

function module.ValidateRigWithIK(rig: Model) : boolean
	if not module.ValidateRig(rig) then return false end
	if not rig:HasTag(IKRig_Tag) then return false end
	return true
end

function module.GetRigType(rig: Model) : ("R6" | "R15")?
	if not module.ValidateRig(rig) then return nil end
	if module.IsBoneRig(rig) then
		return "R6"
	end

	local humanoid = rig:FindFirstChildOfClass("Humanoid")
	if humanoid.RigType == Enum.HumanoidRigType.R6 then
		return "R6"
	elseif humanoid.RigType == Enum.HumanoidRigType.R15 then
		return "R15"
	end
	return nil
end


----// RIG INFO
function module.GetLegs(rig: Model)
	local rigType = module.GetRigType(rig)
	assert(rigType, "Invalid Rig")

	if module.IsBoneRig(rig) then
		return BoneLegAdapter.GetSideParts(rig)
	end

	if rigType == "R6" then
		return { LeftLeg = rig["Left Leg"], RightLeg = rig["Right Leg"] }
	else
		return {
			LeftUpperLeg = rig.LeftUpperLeg,
			LeftLowerLeg = rig.LeftLowerLeg,
			LeftFoot = rig.LeftFoot,
			RightUpperLeg = rig.RightUpperLeg,
			RightLowerLeg = rig.RightLowerLeg,
			RightFoot = rig.RightFoot,
		}
	end
end

function module.GetIkJointsR6(rig: Model, separate: boolean?)
	if isBoneRig(rig) then
		return BoneLegAdapter.GetIkJoints(rig, separate)
	end

	-- ... keep the original body of this function ...

function module.GetIkJointsR15(rig: Model, separate: boolean?)
	
	local rootPart = rig.HumanoidRootPart :: BasePart
	local lTorso = rig:FindFirstChild("LowerTorso")
	local legs, atts = module.GetLegs(rig), module.GetIkAttachments(rig)
	
	local f = R15ratioY
	
	local root = {
		C0 = atts.RootC0 and atts.RootC0.CFrame or CFrame.new(0,-rootPart.Size.Y/2,0),
		C1 = atts.RootC1 and atts.RootC1.CFrame or CFrame.new(0, -lTorso.Size.Y/2, 0)
	}
	local left, right = {}, {}
	
	left.Hip = {
		C0 = atts.LeftHipC0 and atts.LeftHipC0.CFrame or CFrame.new(-lTorso.Size.X/4, -lTorso.Size.Y * f.Hip.C0, 0),
		C1 = atts.LeftHipC1 and atts.LeftHipC1.CFrame or CFrame.new(0, legs.LeftUpperLeg.Size.Y * f.Hip.C1, 0)
	}
	left.Knee = {
		C0 = atts.LeftKneeC0 and atts.LeftKneeC0.CFrame or CFrame.new(0, -legs.LeftUpperLeg.Size.Y * f.Knee.C0, 0),
		C1 = atts.LeftKneeC1 and atts.LeftKneeC1.CFrame or CFrame.new(0, legs.LeftLowerLeg.Size.Y * f.Knee.C1, 0)
	}
	left.Ankle = {
		C0 = atts.LeftAnkleC0 and atts.LeftAnkleC0.CFrame or CFrame.new(0, -legs.LeftLowerLeg.Size.Y * f.Ankle.C0, 0),
		C1 = atts.LeftAnkleC1 and atts.LeftAnkleC1.CFrame or CFrame.new(0, legs.LeftFoot.Size.Y * f.Ankle.C1, 0)
	}
	
	right.Hip = {
		C0 = atts.RightHipC0 and atts.RightHipC0.CFrame or CFrame.new(lTorso.Size.X/4, -lTorso.Size.Y * f.Hip.C0, 0),
		C1 = atts.RightHipC1 and atts.RightHipC1.CFrame or CFrame.new(0, legs.RightUpperLeg.Size.Y * f.Hip.C1, 0)
	}
	right.Knee = {
		C0 = atts.RightKneeC0 and atts.RightKneeC0.CFrame or CFrame.new(0, -legs.RightUpperLeg.Size.Y * f.Knee.C0, 0),
		C1 = atts.RightKneeC1 and atts.RightKneeC1.CFrame or CFrame.new(0, legs.RightLowerLeg.Size.Y * f.Knee.C1, 0)
	}
	right.Ankle = {
		C0 = atts.RightAnkleC0 and atts.RightAnkleC0.CFrame or CFrame.new(0, -legs.RightLowerLeg.Size.Y * f.Ankle.C0, 0),
		C1 = atts.RightAnkleC1 and atts.RightAnkleC1.CFrame or CFrame.new(0, legs.RightFoot.Size.Y * f.Ankle.C1, 0)
	}
	
	if not separate then
		local merged = {Root = root}
		for i, v in right do merged["Right"..i] = v end
		for i, v in left do merged["Left"..i] = v end
		return merged
	end
	
	return left, right, root
end



----/// Calculate R15 IK Controls Positions for IKifying a rig
function module.CalcR15IkControlsCframes(rig: Model)
	local cf, size = rig:GetBoundingBox()
	local rootPart = rig.HumanoidRootPart :: BasePart
	local lTorso = rig:FindFirstChild("LowerTorso") :: BasePart
	local humanoid = rig:FindFirstChildWhichIsA("Humanoid")
	
	local lJoints, rJoints, rootJoint = module.GetIkJointsR15(rig, true)
	local rigScale = module.GetR15RigScale(rig)
	
	local rootOffsetFromGround = (cf * CFrame.new(0,-size.Y/2,0)):ToObjectSpace(rootPart.CFrame).Y
	
	local rFootCf = rootPart.CFrame:ToObjectSpace(rig.RightFoot.CFrame)
	local lFootCf = rootPart.CFrame:ToObjectSpace(rig.LeftFoot.CFrame)
	
	return {
		HeightScale = rigScale.Y,
		RootController = {
			C0 = CFrame.new(), C1 = CFrame.new(0, rootOffsetFromGround - 0.025, 0)
		},
		Root = {
			C0 = CFrame.new(0, rootOffsetFromGround - rigScale.Y - 0.025, 0),
			C1 = rootJoint.C1
		},
		RightLegRoot = {
			C0 = rJoints.Hip.C0, C1 = CFrame.new()
		},
		LeftLegRoot = {
			C0 = lJoints.Hip.C0, C1 = CFrame.new()
		},
		RightLegTarget = {
			C0 = CFrame.new(rFootCf.Position.X, 0.175 * rigScale.Y - 0.025, rFootCf.Position.Z), C1 = CFrame.new()
		},
		LeftLegTarget = {
			C0 = CFrame.new(lFootCf.Position.X, 0.175 * rigScale.Y - 0.025, lFootCf.Position.Z), C1 = CFrame.new()
		},
	}
end

----/// R15 Rig Scale
function module.GetR15RigScale(rig: Model)
	return rig.HumanoidRootPart.Size * Vector3.new(0.5, 0.5, 1)
end

function module.CalcLegLengthR15(rig: Model)
	local lJoints, rJoints = module.GetIkJointsR15(rig, true)
	local rFoot, lFoot = rig["RightFoot"], rig["LeftFoot"]
	
	local leftDir, rightDir = module.GetLegIkEndPointsDirectionR15(rig)
	
	local Right, Left = {}, {}

	Right.L1 = (rJoints.Hip.C1.Position - rJoints.Knee.C0.Position):Dot(rightDir)
	Right.L2 = (rJoints.Knee.C1.Position - rJoints.Ankle.C0.Position):Dot(rightDir)
		+ rJoints.Ankle.C1.Position:Dot(rightDir) + rFoot.Size.Y/2
	Right.Foot = rFoot.Size.Y

	Left.L1 = (lJoints.Hip.C1.Position - lJoints.Knee.C0.Position):Dot(leftDir)
	Left.L2 = (lJoints.Knee.C1.Position - lJoints.Ankle.C0.Position):Dot(leftDir)
		+ lJoints.Ankle.C1.Position:Dot(leftDir) + lFoot.Size.Y/2
	Left.Foot = lFoot.Size.Y
	
	return Left, Right
end

function module.GetLegIkEndPointsDirectionR15(rig: Model)
	local lJoints, rJoints = module.GetIkJointsR15(rig, true)
	
	local rTargetPos = (rJoints.Hip.C0 * rJoints.Hip.C1:Inverse() 
		* rJoints.Knee.C0 * rJoints.Knee.C1:Inverse()
		* rJoints.Ankle.C0 * rJoints.Ankle.C1:Inverse()
		* CFrame.new(0, -rig.RightFoot.Size.Y/2, 0)).Position
	local rRootPos = rJoints.Hip.C0.Position
	
	local lTargetPos = (lJoints.Hip.C0 * lJoints.Hip.C1:Inverse()
		* lJoints.Knee.C0 * lJoints.Knee.C1:Inverse()
		* lJoints.Ankle.C0 * lJoints.Ankle.C1:Inverse()
		* CFrame.new(0, -rig.LeftFoot.Size.Y/2, 0)).Position
	local lRootPos = lJoints.Hip.C0.Position
	
	
	local leftDirection = SafeUnit(lRootPos - lTargetPos)
	local rightDirection = SafeUnit(rRootPos - rTargetPos)
	
	return leftDirection, rightDirection
end

----/// IK Controls
function module.CreateIKControls(rig: Model)
	local rigScale = module.GetR15RigScale(rig)
	local _, boxSize = rig:GetBoundingBox()
	local maxSize = math.max(boxSize.X, boxSize.Z)
	
	for _, v in IK_Controls:GetChildren() do
		local control = v:Clone()
		if v.Name == "RootController" then
			control.Size *= Vector3.new(maxSize/4, 1, maxSize/4)
		else
			control.Size *= rigScale
		end
		control.Parent = rig
	end
end

function module.GetIKControls(rig: Model)
	local controls = {}
	for _, v in rig:GetChildren() do
		if IK_Controls:FindFirstChild(v.Name) then
			controls[v.Name] = v
		end
	end
	return controls
end

function module.RemoveIKControls(rig: Model)
	for _, v in rig:GetChildren() do
		if IK_Controls:FindFirstChild(v.Name) then
			v:Destroy()
		end
	end
end
