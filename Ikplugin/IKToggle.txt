local function getRig()
	for _, item in SelectionService:Get() do
		if item:IsA("Model") and item:FindFirstChildOfClass("Humanoid") then
			return item
		end

		local ancestor = item:FindFirstAncestorOfClass("Model")
		while ancestor do
			if ancestor:FindFirstChildOfClass("Humanoid") then
				return ancestor
			end
			ancestor = ancestor:FindFirstAncestorOfClass("Model")
		end
	end

	if state.rig and state.rig.Parent then
		return state.rig
	end

	local fallback = workspace:FindFirstChild("Rig")
	if fallback and fallback:IsA("Model") and fallback:FindFirstChildOfClass("Humanoid") then
		return fallback
	end

	return nil
end

	end
end

local function solveAll(rig)
	if SpineAdapter then
		SpineAdapter.SolveRig(rig)
	end
	Adapter.SolveRig(rig)
	if Attach then
		Attach.SolveRig(rig)
	end
end

-- ============================================================
-- Запись поз костей
-- ============================================================
local function cframesDiffer(a, b)
	if not a or not b then
		return true
	end

	local ac = { a:GetComponents() }
	local bc = { b:GetComponents() }

	for index = 1, 12 do
		if math.abs(ac[index] - bc[index]) > 0.0005 then
			return true
		end
	end

	return false
end

local function samplesDiffer(a, b)
	if not a or not b then
		return true
	end

	for name, value in a do
		if cframesDiffer(value, b[name]) then
			return true
		end
	end

	return false
end

-- Сэмпл конечностей + спины + головы
local function getFullSample(rig)
	local sample = Adapter.GetSample(rig)

	if SpineAdapter then
		for name, value in SpineAdapter.GetSample(rig) do
			sample[name] = value
		end
	end

	if Attach then
		for name, value in Attach.GetSample(rig) do
			sample[name] = value
		end
	end

	return sample
end

local function recordSample(rig)
	if not recorder.active then
		return
	end

	local elapsed = os.clock() - recorder.startClock

	if elapsed >= RECORD_LIMIT then
		recorder.active = false
		recordButton:SetActive(false)
		warn("[IK Toggle] лимит записи " .. RECORD_LIMIT .. " сек")
		return
	end

	if elapsed - recorder.lastSample < (1 / RECORD_FPS) - 0.0001 then
		return
	end

	recorder.lastSample = elapsed

	table.insert(recorder.samples, {
		time = elapsed,
		bones = getFullSample(rig),
	})
end

local function startRecording()
	if not state.active or not state.rig then
		warn("[IK Toggle] сначала включи Toggle IK")
		return
	end

	solveAll(state.rig)

	recorder.active = true
	recorder.startClock = os.clock()
	recorder.lastSample = 0
	recorder.samples = {
		{
			time = 0,
			bones = getFullSample(state.rig),
		},
	}

	recordButton:SetActive(true)

	print("[IK Toggle] запись началась.")
	print("  Теперь нажми Play в MoonAnimator и проиграй анимацию таргетов.")
	print("  После проигрывания нажми Record IK ещё раз и Bake IK.")
end

local function stopRecording()
	if not recorder.active then
		return
	end

	recorder.active = false
	recordButton:SetActive(false)

	print("[IK Toggle] запись остановлена. Сэмплов: " .. #recorder.samples)
end

-- имена цепочки Part0 -> ... -> Part1 по иерархии Motor6D (от корня рига, без самого корня)
local function motorChainNames(rig, motor, rootName)
	local names = { motor.Part1.Name }
	local current = motor.Part0
	local guard = 0

	while current and guard < 32 do
		guard += 1

		if current.Name == rootName then
			return names
		end

		table.insert(names, 1, current.Name)

		local parentMotor = nil
		for _, descendant in rig:GetDescendants() do
			if descendant:IsA("Motor6D") and descendant.Part1 == current then
				parentMotor = descendant
				break
			end
		end

		if not parentMotor then
			break
		end
		current = parentMotor.Part0
	end

	-- цепочка до корня не найдена
	return { motor.Part1.Name }
end

local function structuralPose(parent, poses, name)
	local pose = poses[name]
	if not pose then
		pose = Instance.new("Pose")
		pose.Name = name
		pose.Weight = 0
		pose.CFrame = CFrame.new()
		pose.Parent = parent
		poses[name] = pose
	end
	return pose
end

local motorChainCache = {}

addMotorPose = function(rootPose, poses, motor, value)
	if not motor.Part0 or not motor.Part1 then
		return
	end

	local names = motorChainCache[motor]
	if not names then
		names = motorChainNames(state.rig, motor, rootPose.Name)
		motorChainCache[motor] = names
	end

	local parent = rootPose
	for index = 1, #names - 1 do
		parent = structuralPose(parent, poses, names[index])
	end

	local name = names[#names]
	local pose = poses[name]
	if not pose then
		pose = Instance.new("Pose")
		pose.Name = name
		pose.Parent = parent
		poses[name] = pose
	end
	pose.Weight = 1
	pose.CFrame = value
end

addBonePose = function(rootPose, poses, bone, value)
	local names = {}
	local current = bone.Parent
	while current and current:IsA("Bone") do
		table.insert(names, 1, current.Name)
		current = current.Parent
	end

	local parent = rootPose
	for _, name in names do
		parent = structuralPose(parent, poses, name)
	end

	local pose = Instance.new("Pose")
	pose.Name = bone.Name
	pose.Weight = 1
	pose.CFrame = value
	pose.Parent = parent
	poses[bone.Name] = pose
end

local function bakeRecording()
	stopRecording()

	if #recorder.samples < 2 then
		warn("[IK Toggle] запись пустая: Toggle IK -> Record IK -> Play в Moon -> Record IK -> Bake IK")
		return
	end

	if not state.rig or not state.rig.Parent then
		warn("[IK Toggle] риг не найден")
		return
	end

	local sides = Adapter.GetSides(state.rig)
	if #sides == 0 then
		warn("[IK Toggle] адаптер не нашёл ноги")
		return
	end

	local sequence = Instance.new("KeyframeSequence")
	sequence.Name = "BoneIK_" .. state.rig.Name
		.. "_" .. tostring(math.floor(os.clock() * 1000))
	sequence.Loop = false
	sequence.Priority = Enum.AnimationPriority.Action

	local previous = nil
	local keysAdded = 0

	for index, sample in recorder.samples do
		local isLast = index == #recorder.samples
		local differs = previous == nil or samplesDiffer(previous, sample.bones)

		if differs or isLast then
			local keyframe = Instance.new("Keyframe")
			keyframe.Time = sample.time

			local rootPose = Instance.new("Pose")
			rootPose.Name = "HumanoidRootPart"
			rootPose.Weight = 1
			rootPose.CFrame = CFrame.new()
			rootPose.Parent = keyframe

			for _, side in sides do
				local thighPose = Instance.new("Pose")
				thighPose.Name = side.Thigh.Name
				thighPose.Weight = 1
				thighPose.CFrame = sample.bones[side.Thigh.Name] or CFrame.new()
				thighPose.Parent = rootPose

				local kneePose = Instance.new("Pose")
				kneePose.Name = side.Knee.Name
				kneePose.Weight = 1
				kneePose.CFrame = sample.bones[side.Knee.Name] or CFrame.new()
				kneePose.Parent = thighPose
			end

			-- Спина: цепочка TorsoBoneRoot -> ... -> TorsoBoneNeck (вложенные позы)
			local poses = {}

			if SpineAdapter then
				local parentPose = rootPose

				for _, bone in SpineAdapter.GetBones(state.rig) do
					local bonePose = Instance.new("Pose")
					bonePose.Name = bone.Name
					bonePose.Weight = 1
					bonePose.CFrame = sample.bones[bone.Name] or CFrame.new()
					bonePose.Parent = parentPose
					poses[bone.Name] = bonePose
					parentPose = bonePose
				end

				-- Голова: поза OffsetPart (Motor6D.Transform).
				-- Pose должна лежать в иерархии Motor6D (Part0 -> Part1), а не прямо под корнем.
				local headMotor = SpineAdapter.GetHeadMotor(state.rig)
				if headMotor and sample.bones["__HeadMotor"] then
					addMotorPose(rootPose, poses, headMotor, sample.bones["__HeadMotor"])
				end
			end

			-- Подключённые объекты (Add IK Handle): кости и Motor6D
			if Attach then
				for _, info in Attach.GetPoseInfo(state.rig) do
					local value = sample.bones[info.Key]
					if value then
						if info.Kind == "Motor" then
							addMotorPose(rootPose, poses, info.Motor, value)
						elseif info.Kind == "Bone" and not poses[info.Bone.Name] then
							addBonePose(rootPose, poses, info.Bone, value)
						end
					end
				end
			end

			keyframe.Parent = sequence
			keysAdded += 1
		end

		previous = sample.bones
	end

	local folder = ReplicatedStorage:FindFirstChild("BoneIK_Recordings")
	if not folder then
		folder = Instance.new("Folder")
		folder.Name = "BoneIK_Recordings"
		folder.Parent = ReplicatedStorage
	end

	sequence.Parent = folder

	pcall(function()
		SelectionService:Set({ sequence })
	end)

	print("==============================================")
	print("[IK Toggle] сохранено: " .. sequence:GetFullName())
	print(string.format(
		"  ключей: %d, длительность: %.2f сек",
		keysAdded,
		recorder.samples[#recorder.samples].time
		))
	print("  Импорт в MoonAnimator как вторая анимация:")
	print("  Save to Roblox -> Animation ID -> Moon Import -> Animation ID.")
	print("==============================================")
end

-- ============================================================
-- Live IK: PreRender + Heartbeat
-- ============================================================
local function tickSolve()
	if not state.active then
		return
	end

	if not state.rig or not state.rig.Parent then
		return
	end

	-- Сначала спина и голова (считают дельты), потом ноги/руки, потом подключённые объекты
	solveAll(state.rig)
	recordSample(state.rig)
end

local function connectSolvers()
	state.heartbeat = RunService.Heartbeat:Connect(tickSolve)
	state.render = nil

	local preRenderOk = pcall(function()
		state.render = RunService.PreRender:Connect(tickSolve)
	end)

	if not preRenderOk or not state.render then
		pcall(function()
			state.render = RunService.RenderStepped:Connect(tickSolve)
		end)
	end

	print("[IK Toggle] solvers: Heartbeat"
		.. (state.render and " + PreRender/RenderStepped" or ""))
end

local function disconnectSolvers()
	if state.heartbeat then
		state.heartbeat:Disconnect()
		state.heartbeat = nil
	end

	if state.render then
		state.render:Disconnect()
		state.render = nil
	end
end

-- ============================================================
-- IK вкл/выкл
-- ============================================================
local function stopIK(restore)
	disconnectSolvers()
	stopRecording()

	if restore ~= false and state.rig and state.rig.Parent then
		pcall(Adapter.Restore, state.rig)
		if SpineAdapter then
			pcall(SpineAdapter.Restore, state.rig)
		end
		if Attach then
			pcall(Attach.Restore, state.rig)
		end
	end

	state.active = false
	updateFrames()
	setToggleActive(false)
end

local function startIK(rig)
	stopIK(true)

	local ok, message = Adapter.EnsureControls(rig)

	if not ok then
		warn("[IK Toggle] " .. tostring(message))
		return false
	end

	Adapter.Bind(rig)

	-- Спина + OffsetPart головы (необязательно: если костей нет, работает как раньше)
	if SpineAdapter then
		local spineOk, spineMessage = SpineAdapter.EnsureControls(rig)
		if not spineOk then
			warn("[IK Toggle] спина отключена: " .. tostring(spineMessage))
		end
	end

	local sides = Adapter.GetSides(rig)
	if #sides == 0 then
		warn("[IK Toggle] адаптер не нашёл ноги")
		return false
	end

	-- Подключённые объекты (хэндлы <Имя>_IKHandle)
	if Attach then
		Attach.Rebind(rig)
	end

	state.rig = rig
	state.active = true

	solveAll(rig)
	connectSolvers()

	updateFrames()
	setToggleActive(true)

	print("[IK Toggle] live IK включён: " .. rig:GetFullName())

	for _, side in sides do
		print(string.format(
			"  %s / %s -> %s",
			side.Thigh.Name,
			side.Knee.Name,
			side.Target and side.Target.Name or "MISSING"
			))
	end

	print("  Управление: двигай таргеты, поулы и *RootControl в MoonAnimator.")
	print("  Корни: RightLegRootControl / LeftLegRootControl / RightArmRootControl / LeftArmRootControl.")
	print("  Спина: SpineBaseControl (бёдра), SpineLumbarControl (поясница), SpineTarget (шея), HeadControl (OffsetPart).")
	print("  Ноги считаются от TorsoBoneRoot, руки — от TorsoBone.003.")
	print("  Кости в Moon не добавляй, иначе Moon перезапишет IK.")

	return true
end

-- ============================================================
-- Выделение таргетов для MoonAnimator
-- ============================================================
local function selectIkParts()
	if not Adapter then
		warn("[IK Toggle] Utils/BoneLegAdapter не загружен")
		return
	end

	local rig = getRig()
	if not rig then
		warn("[IK Toggle] выдели риг")
		return
	end

	if not state.active then
		local ok = startIK(rig)
		if not ok then
			return
		end
	end

	-- таргеты + поулы + RootControl для всех цепочек (ноги и руки)
	local targets = Adapter.GetTargets(rig)

	if SpineAdapter then
		for _, part in SpineAdapter.GetTargets(rig) do
			table.insert(targets, part)
		end
	end

	if Attach then
		for _, part in Attach.GetTargets(rig) do
			table.insert(targets, part)
		end
	end

	if #targets == 0 then
		warn("[IK Toggle] таргеты не созданы")
		return
	end

	SelectionService:Set(targets)

	print("==============================================")
	print("[IK Toggle] выделены таргеты IK:")
	for _, part in targets do
		print("  " .. part:GetFullName())
	end
	print("")
	print("  Дальше в MoonAnimator:")
	print("  1. Нажми Add Item / 9 и добавь все выделенные контролы.")
	print("  2. У каждого появится CFrame-трек.")
	print("  3. Двигай таргеты, поулы и *RootControl, ставь ключи (+).")
	print("     *RootControl двигает корень кости (RightBoneRoot и т.д.).")
	print("  4. Кости и сам Rig в Moon не добавляй.")
	print("  5. Во время Play плагин сам держит IK-позу.")
	print("==============================================")
end

local function selectRootControls()
	local rig = getRig()
	if not rig then
		warn("[IK Toggle] выдели риг")
		return
	end

	if not state.active then
		if not startIK(rig) then
			return
		end
	end

	local roots = Adapter.GetRootControls(rig)

		warn("[IK Toggle] RootControl не найдены (проверь Config.RootControlsEnabled)")
		return
	end

	SelectionService:Set(roots)

	print("[IK Toggle] выделены хэндлы корней:")
	for _, part in roots do
		print("  " .. part:GetFullName())
	end
	print("  Двигай их Move-тулом или добавь в MoonAnimator и ставь ключи.")
end

local function resetSpineControls()
	if not SpineAdapter then
		return
	end

	local rig = getRig()
	if not rig then
		warn("[IK Toggle] выдели риг")
		return
	end

	local count = SpineAdapter.ResetControls(rig)
	if Attach then
		count += Attach.ResetControls(rig)
	end
	print("[IK Toggle] контролов спины/головы/подключённых сброшено: " .. tostring(count))
end

local function resetRootControls()
	local rig = getRig()
	if not rig then
		warn("[IK Toggle] выдели риг")
		return
	end

	local count = Adapter.ResetRootControls(rig)
	print("[IK Toggle] хэндлов корней сброшено: " .. tostring(count))
end

-- ============================================================
-- Подключение чего угодно к IK
-- ============================================================
local function addHandles()
	if not Attach then
		warn("[IK Toggle] Utils/IKAttach не загружен — добавь модуль IKAttach в папку Utils")
		return
	end

	local rig = getRig()
	if not rig then
		warn("[IK Toggle] выдели объекты внутри рига (Model с Humanoid)")
		return
	end

	local items = {}
	for _, item in SelectionService:Get() do
		if item ~= rig then
			table.insert(items, item)
		end
	end

	if #items == 0 then
		warn("[IK Toggle] выдели детали / модели / кости рига, которые хочешь подключить к IK")
		return
	end

	if not (state.active and state.rig == rig) then
		if not startIK(rig) then
			return
		end
	end

	local created, messages = Attach.Add(rig, items)

	for _, message in messages do
		warn("[IK Toggle] пропущено: " .. message)
	end

	if #created == 0 then
		return
	end

	SelectionService:Set(created)

	print("==============================================")
	print("[IK Toggle] подключено к IK: " .. #created)
	for _, handle in created do
		local value = handle:FindFirstChild("IKTarget")
		print("  " .. handle.Name .. "  ->  " .. (value and value.Value and value.Value:GetFullName() or "?"))
	end
	print("  Двигай хэндлы или добавь их в MoonAnimator (Add Item / 9).")
	print("  Атрибуты хэндла: FollowBone (за какой костью следовать), Mode (Auto/Motor/Direct), Enabled.")
	print("==============================================")
end

local function removeHandles()
	if not Attach then
		return
	end

	local rig = getRig()
	if not rig then
		warn("[IK Toggle] выдели хэндл или подключённый объект")
		return
	end

	local items = SelectionService:Get()
	local count = Attach.Remove(rig, items)
	print("[IK Toggle] хэндлов удалено: " .. tostring(count))
end

local function diagnose()
	local rig = getRig()
	if not rig then
		warn("[IK Toggle] выдели риг")
		return
	end

	print("==============================================")
	print("[IK Toggle] ДИАГНОСТИКА " .. rig:GetFullName() .. " (IK " .. (state.active and "включён" or "выключен") .. ")")

	if SpineAdapter then
		local driver = SpineAdapter.GetHeadDriver(rig)
		if driver then
			print("  [Голова]")
			for _, line in driver:Describe() do
				print("  " .. line)
			end
		else
			print("  [Голова] драйвер не создан (нет OffsetPart / IKAttach / ещё не нажат Toggle IK)")
		end
	end

	if Attach then
		print("  [Подключённые объекты]")
		Attach.Diagnose(rig)
	end
	print("==============================================")
end
